Q42. Explain the buyTrademark function step by step.
ANSWER:
```solidity
function buyTrademark(uint256 listingId) external payable nonReentrant {
    // 1. Get listing from storage
    Listing storage listing = listings[listingId];
    
    // 2. Validate listing is active and not expired
    require(listing.active, "Listing is not active");
    require(!listing.isLicense, "This is a license listing");
    require(listing.expiresAt == 0 || listing.expiresAt > block.timestamp);
    
    // 3. Validate payment
    require(msg.value == listing.price, "Incorrect payment amount");
    require(msg.sender != listing.seller, "Cannot buy your own listing");
    
    // 4. Calculate royalty using EIP-2981
    (address royaltyReceiver, uint256 royaltyAmount) = 
        trademarkNFT.royaltyInfo(listing.tokenId, listing.price);
    
    // 5. Calculate marketplace fee
    uint256 marketplaceFeeAmount = (listing.price * marketplaceFee) / 10000;
    
    // 6. Calculate seller amount
    uint256 sellerAmount = listing.price - royaltyAmount - marketplaceFeeAmount;
    
    // 7. Mark listing inactive
    listing.active = false;
    
    // 8. Transfer NFT to buyer
    trademarkNFT.safeTransferFrom(listing.seller, msg.sender, listing.tokenId);
    
    // 9. Distribute payments
    if (royaltyAmount > 0 && royaltyReceiver != listing.seller) {
        payable(royaltyReceiver).transfer(royaltyAmount);
    } else {
        sellerAmount += royaltyAmount;
    }
    payable(owner()).transfer(marketplaceFeeAmount);
    payable(listing.seller).transfer(sellerAmount);
    
    // 10. Emit event
    emit TrademarkSold(...);
}
```

```typescript
  // State management
  const [account, setAccount] = useState<string | null>(null);

    // Event listeners
    if (window.ethereum) {
      window.ethereum.on('accountsChanged', handleAccountsChanged);
      window.ethereum.on('chainChanged', handleChainChanged);
    }
    
    // Cleanup
    return () => {
      if (window.ethereum) {
        window.ethereum.removeListener('accountsChanged', handleAccountsChanged);
        window.ethereum.removeListener('chainChanged', handleChainChanged);
      }
    };
  }, []);

  // Connect function
  const connect = async () => {
    setIsLoading(true);
    const provider = new ethers.BrowserProvider(window.ethereum);
    await provider.send('eth_requestAccounts', []);
    const signer = await provider.getSigner();
    const address = await signer.getAddress();
    setAccount(address);
    setIsConnected(true);
    setIsLoading(false);
  };

  return (
    <Web3Context.Provider value={{ account, chainId, isConnected, connect, ... }}>
      {children}
    </Web3Context.Provider>
  );
}
```

Q46. What is gas optimization and how did you implement it?
ANSWER:
GAS OPTIMIZATION TECHNIQUES:
- Use uint256 instead of smaller uints (except for packing)
- Pack struct variables efficiently
- Use immutable for constants set in constructor
- Avoid unnecessary storage writes
- Use memory instead of storage where possible
- Batch operations when possible
- Use events instead of storage for historical data
- Short-circuit boolean operations

EXAMPLE:
```solidity
// Expensive - multiple storage writes
listing.active = false;
listing.soldAt = block.timestamp;
listing.buyer = msg.sender;

// Better - single storage write with struct
listings[listingId] = Listing({
    ...listing,
    active: false,
    soldAt: block.timestamp,
    buyer: msg.sender
});
```

Q47. Explain the EIP-2981 royalty standard.
ANSWER:
EIP-2981 provides standard way to retrieve royalty payment information:

```solidity
function royaltyInfo(uint256 tokenId, uint256 salePrice) 
    external view returns (address receiver, uint256 royaltyAmount) {
    // Get royalty percentage for token
    uint96 royaltyBps = _tokenRoyalties[tokenId];
    
    // Calculate royalty amount (basis points)
    royaltyAmount = (salePrice * royaltyBps) / 10000;
    
    // Return creator address and amount
    receiver = _tokenCreators[tokenId];
}
```

Benefits:
- Standardized across marketplaces
- Automatic royalty calculation
- Creator gets paid on secondary sales
- Basis points (1/100th of percent) for precision

Q48. What is the difference between memory and storage in Solidity?
ANSWER:
STORAGE:
- Persistent data stored on blockchain
- Expensive (costs gas)
- State variables are storage by default
- Modifications are permanent

MEMORY:
- Temporary data during function execution
- Cheaper than storage
- Cleared after function completes
- Used for function parameters and local variables

EXAMPLE:
```solidity
// Storage - persistent
Listing storage listing = listings[listingId];
listing.active = false; // Modifies blockchain state

// Memory - temporary
Listing memory tempListing = listings[listingId];
tempListing.active = false; // Only in memory, not saved
```

```typescript
// Verify wallet signature
export async function verifyWalletSignature(
  address: string, 
  signature: string, 
  message: string
): Promise<boolean> {
  try {
    const recoveredAddress = ethers.verifyMessage(message, signature);
    return recoveredAddress.toLowerCase() === address.toLowerCase();
  } catch {
    return false;
  }
}

// Middleware for protected routes
export function requireAuth(handler) {
  return async (req, res) => {
    const { address, signature } = req.headers;
    
    if (!address || !signature) {
      return res.status(401).json({ error: 'Unauthorized' });
    }
    
    const isValid = await verifyWalletSignature(address, signature, 'Auth message');
    
    if (!isValid) {
      return res.status(403).json({ error: 'Invalid signature' });
    }
    
    return handler(req, res);
  };
}
```

Q54. How do you handle pagination in your API?
ANSWER:
```typescript
export default async function handler(req, res) {
  const { page = 1, limit = 20, category, verified } = req.query;
  
  let query = db.collection('trademarks');
  
  // Apply filters
  if (category) {
    query = query.where('category', '==', category);
  }
  if (verified === 'true') {
    query = query.where('verificationStatus', '==', 'verified');
  }
  
  // Pagination
  const offset = (parseInt(page) - 1) * parseInt(limit);
  query = query.orderBy('createdAt', 'desc')
               .limit(parseInt(limit))
               .offset(offset);
  
  const snapshot = await query.get();
  const trademarks = snapshot.docs.map(doc => doc.data());
  
  // Get total count
  const totalSnapshot = await db.collection('trademarks').count().get();
  const total = totalSnapshot.data().count;
  
  return res.json({
    data: trademarks,
    pagination: {
      page: parseInt(page),
      limit: parseInt(limit),
      total,
      totalPages: Math.ceil(total / parseInt(limit))
    }
  });
}
```
