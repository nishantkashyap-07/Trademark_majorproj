```typescript
const switchNetwork = async () => {
  try {
    await window.ethereum.request({
      method: 'wallet_switchEthereumChain',
      params: [{ chainId: '0x89' }], // Polygon mainnet
    });
  } catch (error) {
    if (error.code === 4902) {
      // Network not added, add it
      await window.ethereum.request({
        method: 'wallet_addEthereumChain',
        params: [{
          chainId: '0x89',
          chainName: 'Polygon Mainnet',
          rpcUrls: ['https://polygon-rpc.com'],
          nativeCurrency: { name: 'MATIC', symbol: 'MATIC', decimals: 18 },
          blockExplorerUrls: ['https://polygonscan.com']
        }]
      });
    }
  }
};
```

Q83. What if transaction fails?
ANSWER:
COMMON CAUSES:
1. Insufficient gas
2. User rejection
3. Network congestion
4. Contract revert (validation failed)

HANDLING:
```typescript
try {
  const tx = await contract.buyTrademark(listingId, { value: price });
  await tx.wait();
} catch (error) {
  if (error.code === 4001) {
    showError('Transaction rejected by user');
  } else if (error.code === 'INSUFFICIENT_FUNDS') {
    showError('Insufficient balance');
  } else if (error.reason) {
    showError(`Transaction failed: ${error.reason}`);
  } else {
    showError('Transaction failed. Please try again.');
  }
}
```

Q84. What if IPFS content doesn't load?
ANSWER:
FALLBACK MECHANISM:
```typescript
const IPFS_GATEWAYS = [
  'https://ipfs.io/ipfs/',
  'https://gateway.pinata.cloud/ipfs/',
  'https://cloudflare-ipfs.com/ipfs/',
  'https://dweb.link/ipfs/'
];

async function fetchFromIPFS(hash) {
  for (const gateway of IPFS_GATEWAYS) {
    try {
      const response = await fetch(gateway + hash, { timeout: 10000 });
      if (response.ok) return response;
    } catch (error) {
      continue; // Try next gateway
    }
  }
  throw new Error('Failed to load from IPFS');
}
```

Q86. How do you optimize smart contract gas usage?
ANSWER:
1. USE EVENTS INSTEAD OF STORAGE:
   ```solidity
   // Expensive
   mapping(uint256 => string[]) public history;
   
   // Cheaper
   event HistoryUpdated(uint256 indexed id, string data);
   ```

2. PACK VARIABLES:
   ```solidity
   // Expensive - 3 storage slots
   uint256 a;
   uint8 b;
   uint256 c;
   
   // Cheaper - 2 storage slots
   uint256 a;
   uint256 c;
   uint8 b;
   ```

3. USE IMMUTABLE:
   ```solidity
   address public immutable trademarkNFT; // Set once, cheaper reads
   ```

4. BATCH OPERATIONS:
   ```solidity
   function batchMint(uint256[] calldata amounts) external {
     // Process multiple in one transaction
   }
   ```
